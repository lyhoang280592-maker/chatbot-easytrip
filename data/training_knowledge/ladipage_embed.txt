<script>
(function() {
  const API_URL = "https://chatbot-easytrip.onrender.com/chat";
  
  const toggleBtn = document.getElementById('et-toggle-btn');
  const chatBox = document.getElementById('et-chat-box');
  const closeBtn = document.getElementById('et-close-btn');
  const inputEl = document.getElementById('et-input');
  const sendBtn = document.getElementById('et-send');
  const bodyEl = document.getElementById('et-body');
  
  const iconChat = document.getElementById('et-icon-chat');
  const iconClose = document.getElementById('et-icon-close');

  // Get or generate a unique user_id for this visitor
  let userId = localStorage.getItem('et_visitor_id');
  if (!userId) {
    userId = 'visitor_' + Math.random().toString(36).substring(2, 11) + '_' + Date.now();
    localStorage.setItem('et_visitor_id', userId);
  }

  let history = [];
  let isOpen = false;
  let isWaitingForResponse = false;
  let lastHistoryLength = 0;

  function toggleChat() {
    isOpen = !isOpen;
    if (isOpen) {
      chatBox.classList.add('et-active');
      iconChat.style.display = 'none';
      iconClose.style.display = 'block';
      setTimeout(() => inputEl.focus(), 300);
    } else {
      chatBox.classList.remove('et-active');
      iconChat.style.display = 'block';
      iconClose.style.display = 'none';
    }
  }

  toggleBtn.addEventListener('click', toggleChat);
  closeBtn.addEventListener('click', () => { if(isOpen) toggleChat(); });

  function formatText(text) {
    if (!text) return '';
    // Replace markdown image ![alt](url) with <img> tag
    let formatted = text.replace(/!\[(.*?)\]\((.*?)\)/g, '<img src="$2" alt="$1" class="chat-image" style="max-width: 100%; border-radius: 8px; margin-top: 10px; display: block;">');
    // Replace markdown link [text](url) with <a> tag
    formatted = formatted.replace(/\[(.*?)\]\((.*?)\)/g, '<a href="$2" target="_blank" style="color: #3b82f6; text-decoration: underline;">$1</a>');
    // Replace markdown bold **text** with <strong>text</strong>
    formatted = formatted.replace(/\*\*(.*?)\*\*/g, '<strong>$1</strong>');
    // Replace markdown italic *text* with <em>text</em>
    formatted = formatted.replace(/\*(.*?)\*/g, '<em>$1</em>');
    // Replace newlines with <br>
    return formatted.replace(/\n/g, '<br>');
  }

  function renderMsg(txt, type) {
    const d = document.createElement('div');
    d.className = `et-msg et-${type}`;
    d.innerHTML = `<div class="et-bubble">${formatText(txt)}</div>`;
    bodyEl.appendChild(d);
    bodyEl.scrollTop = bodyEl.scrollHeight;
  }

  function renderHistory(historyList) {
    bodyEl.innerHTML = '';
    
    // Always render greeting first
    renderMsg("Xin chào! 👋 Mình là chuyên viên tư vấn Visa của EasyTrip. Mình có thể giúp gì cho chuyến đi sắp tới của bạn?", 'bot');
    
    if (historyList && historyList.length > 0) {
      historyList.forEach(msg => {
        const type = msg.role === 'user' ? 'user' : 'bot';
        renderMsg(msg.content, type);
      });
    }
  }

  function showTyping() {
    const d = document.createElement('div');
    d.className = 'et-msg et-bot';
    d.id = 'et-typing';
    d.innerHTML = '<div class="et-bubble et-typing-box"><div class="et-typing"><span></span><span></span><span></span></div></div>';
    bodyEl.appendChild(d);
    bodyEl.scrollTop = bodyEl.scrollHeight;
  }

  function hideTyping() {
    const el = document.getElementById('et-typing');
    if(el) el.remove();
  }

  async function sendMsg() {
    if (isWaitingForResponse) return;
    const txt = inputEl.value.trim();
    if(!txt) return;

    renderMsg(txt, 'user');
    inputEl.value = '';
    history.push({ role: 'user', content: txt });
    showTyping();
    isWaitingForResponse = true;

    try {
      const res = await fetch(API_URL, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ user_id: userId, messages: history })
      });

      if(!res.ok) throw new Error('Network Err');
      
      const data = await res.json();
      hideTyping();
      renderMsg(data.reply, 'bot');
      history.push({ role: 'model', content: data.reply });
      lastHistoryLength += 2;
    } catch (e) {
      hideTyping();
      renderMsg('Đường truyền chậm, xin vui lòng thử lại.', 'bot');
      history.pop();
    } finally {
      isWaitingForResponse = false;
    }
  }

  async function pollHistory() {
    if (isWaitingForResponse) return;
    
    try {
      const res = await fetch(`https://chatbot-easytrip.onrender.com/chat/${userId}/history`);
      if (!res.ok) return;
      
      const data = await res.json();
      if (data.success && data.history) {
        if (data.history.length !== lastHistoryLength) {
          lastHistoryLength = data.history.length;
          
          // Sync history array
          history = [];
          data.history.forEach(msg => {
            history.push({ role: msg.role === 'user' ? 'user' : 'model', content: msg.content });
          });
          
          renderHistory(data.history);
        }
      }
    } catch (e) {
      console.error('Error polling history:', e);
    }
  }

  sendBtn.addEventListener('click', sendMsg);
  inputEl.addEventListener('keypress', (e) => { if(e.key === 'Enter') sendMsg(); });

  // Start polling
  pollHistory();
  setInterval(pollHistory, 3000);
})();
</script>
